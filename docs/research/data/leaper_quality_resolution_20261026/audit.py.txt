ROOT=Path.cwd(); DATA=ROOT/'docs/research/data'; OUT=ROOT/'.local_agent/resolution-97f64b80'
sources={}
def load(folder,name):
 p=DATA/folder/name; b=p.read_bytes(); sources[str(p.relative_to(ROOT)).replace(chr(92),'/')]=hashlib.sha256(b).hexdigest()
 text=gzip.decompress(b).decode() if name.endswith('.gz') else b.decode()
 return [json.loads(v) for v in text.splitlines()] if '.jsonl' in name else json.loads(text)
def key(record):
 # Structural complete-record join, NOT a Core state identity/TT key.
 return hashlib.sha256(json.dumps(record,sort_keys=True,separators=(',',':')).encode()).hexdigest()
def save(name,value):
 (OUT/name).write_text(json.dumps(value,ensure_ascii=False,sort_keys=True,indent=2)+'\n',encoding='utf-8',newline='\n')

# Exact conditional two-sided sign tests. This is a sensitivity illustration,
# NOT a valid p-value/power claim for the frozen, dependent proposed games.
def tail(n,k,p):return sum(math.comb(n,j)*p**j*(1-p)**(n-j) for j in range(k,n+1))
def exact_p(n,k):return min(Fraction(1),2*sum((Fraction(math.comb(n,j),2**n) for j in range(k,n+1)),Fraction(0)))
resolution=[]
for n,unit in [(16,'hypothetical IID decisive games'),(8,'hypothetical independent history/color-pair signs'),(4,'hypothetical independent rule-cluster signs')]:
 threshold=next((k for k in range(n//2+1,n+1) if exact_p(n,k)<=Fraction(1,20)),None)
 # Independently enumerate all fair sign configurations for each upper tail.
 for k in range(n//2+1,n+1):
  count=sum(sum(bits)>=k for bits in itertools.product((0,1),repeat=n))
  assert exact_p(n,k)==min(Fraction(1),Fraction(2*count,2**n))
 target=None
 if threshold:
  lo,hi_p=.5,1.
  for _ in range(60):
   mid=(lo+hi_p)/2
   if tail(n,threshold,mid)<.8:lo=mid
   else:hi_p=mid
  target=(lo+hi_p)/2
 resolution.append({'units':n,'unit':unit,'two_sided_alpha':.05,'minimum_positive_signs_if_all_units_decisive':threshold,'all_positive_exact_p':str(exact_p(n,n)),'positive_rejection_probability_if_sign_p_point8':tail(n,threshold,.8) if threshold else 0.,'sign_p_for_80percent_positive_rejection_no_draws':target,'not_valid_inference_for_actual_panel':True})
assert exact_p(8,7)>Fraction(1,20) and exact_p(8,8)<Fraction(1,20)
assert exact_p(4,4)==Fraction(1,8) and exact_p(16,13)<Fraction(1,20)<exact_p(16,12)

Conditional sign probability curves are not Elo, actual strength power or a fixed effect promise.'})
