        static const char* aux_env = std::getenv("DEV_AUX_EVAL_VALUES");
        static const std::array<int,16> aux_values = [](){
            std::array<int,16> v{}; const char* e=std::getenv("DEV_AUX_EVAL_VALUES");

            for(int c=0;c<2;++c) for(int pt=1;pt<=7;++pt) score+=(c==0?1:-1)*hand_count(pos.hand_of(Color(c)),PieceType(pt))*aux_values[pt];
