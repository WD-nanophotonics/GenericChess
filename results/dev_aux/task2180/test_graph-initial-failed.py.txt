def execute_strategy(s,cert):
 cur=GraphStrategyCursor(s,cert,seconds=3.);actual=copy(s);moves=0
 while actual.result.status is SessionStatus.ONGOING:
  assert moves<96
  choice=cur.choose(actual)if actual.state.position.side_to_move==cert.target else cur.certificate.nodes[cur.index].edges[-1][0]
  cur.commit(actual,choice);moves+=1
  replay=GameSession.replay(actual.compiled,actual.to_record());assert _same_state(replay.state,actual.state)and replay.result==actual.result
 assert actual.result.winner==cert.target;return moves
